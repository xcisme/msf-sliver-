import { ref, computed, onMounted } from 'vue'
import { ElMessage, ElMessageBox } from 'element-plus'
import { apiFunction1, apiFunction2 } from '../api'

// 响应式状态
const data = ref([])

  loading.value = true

    ElMessage.error('加载失败')
  } finally {
    loading.value = false
  }
}

onMounted(() => {
  fetchData()
})
</script>

### API 接口封装规范
在 `src/api/index.js` 中添加新接口函数：
```javascript
// 获取 XXX 列表
export const getXxxList = (params) => {
  return request.get('/api/xxx', { params })
}

// 创建 XXX
export const createXxx = (data) => {
  return request.post('/api/xxx', data)
}

// 删除 XXX
export const deleteXxx = (id) => {
  return request.delete(`/api/xxx/${id}`)
}
```

**重要**：`request` 实例的响应拦截器已处理 `response.data`，所以 API 函数直接返回数据。

### Element Plus 组件使用约定

- 表格：`<el-table :data="list" v-loading="loading">`
- 弹窗：`<el-dialog v-model="visible" title="标题" width="500px">`
- 表单：`<el-form ref="formRef" :model="form" :rules="rules">`
- 标签：`<el-tag :type="getType(value)" size="small">`
- 确认：`ElMessageBox.confirm('确定吗？', '确认', { type: 'warning' })`
- 提示：`ElMessage.success/error/warning/info('消息')`

### 路由导航守卫
- `/` → Login.vue（未登录时）
- `/dashboard` → AppLayout.vue（需登录）
- Token 存储在 `localStorage.getItem('c2_token')`
- 401 时自动清除 token 并跳转登录页

## 接收任务后的工作流程

1. 阅读任务相关的现有前端代码（页面、API 函数、路由）
2. 确认需要对接的后端 API 端点路径和参数格式
3. 如需新接口，先在 `api/index.js` 添加 API 函数（标注后端接口路径）
4. 编写 Vue 组件，使用项目一致的 Element Plus 风格
5. 确保 scoped style 不会与全局样式冲突
6. 预留好 API 接口调用位置，即使后端尚未实现
7. 添加 loading 状态和错误处理
8. 向架构工程师汇报完成情况，标注哪些接口等待后端对接
